db/schema.sql (Neon Postgres + pgvector) 
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "vector";

CREATE TABLE IF NOT EXISTS photos (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  filename VARCHAR(255) NOT NULL,
  title VARCHAR(255) NOT NULL,
  category VARCHAR(64) NOT NULL,
  captured_at TIMESTAMPTZ DEFAULT NOW(),
  location_name VARCHAR(180),
  camera_model VARCHAR(120),
  ai_tags TEXT[] DEFAULT '{}',
  ai_confidence DECIMAL(5,2) DEFAULT 98.40,
  aesthetic_score DECIMAL(4,1) DEFAULT 9.2,
  sharpness_score DECIMAL(5,1) DEFAULT 94.0,
  phash VARCHAR(64) NOT NULL,
  is_blurry BOOLEAN DEFAULT FALSE,
  is_duplicate BOOLEAN DEFAULT FALSE,
  is_favorite BOOLEAN DEFAULT FALSE,
  clip_embedding vector(512)
);
netlify/functions/photos.js (Serverless Handler) 
import { neon } from '@neondatabase/serverless';

export default async (req) => {
  const sql = neon(process.env.DATABASE_URL);
  if (req.method === 'GET') {
    const rows = await sql`
      SELECT * FROM photos
      WHERE is_archived = FALSE
      ORDER BY captured_at DESC LIMIT 200
    `;
    return Response.json({ ok: true, photos: rows });
  }
  if (req.method === 'POST') {
    const b = await req.json();
    const res = await sql`
      INSERT INTO photos (filename, title, category, phash)
      VALUES (${b.filename}, ${b.title}, ${b.category}, ${b.phash})
      RETURNING *
    `;
    return Response.json({ ok: true, photo: res[0] }, { status: 201 });
  }
};
